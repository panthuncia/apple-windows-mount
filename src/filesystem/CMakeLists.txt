find_path(WINFSP_INCLUDE_DIR winfsp/winfsp.h HINTS "C:/Program Files (x86)/WinFsp/inc" "${PROJECT_SOURCE_DIR}/.cache/winfsp-sdk/DYNAMIC/inc" REQUIRED)
find_library(WINFSP_LIBRARY NAMES winfsp-x64 HINTS "C:/Program Files (x86)/WinFsp/lib" "${PROJECT_SOURCE_DIR}/.cache/winfsp-sdk/DYNAMIC/lib" REQUIRED)
add_library(awm_filesystem STATIC names.cpp afc_session.cpp volume.cpp)

target_link_libraries(awm_filesystem PUBLIC awm_device ${WINFSP_LIBRARY} advapi32)
target_compile_options(awm_filesystem PRIVATE /W4 /EHsc /utf-8)
target_compile_definitions(awm_filesystem PUBLIC UNICODE _UNICODE)
add_executable(awm-mount main.cpp)
target_link_libraries(awm-mount PRIVATE awm_filesystem)

install(TARGETS awm-mount RUNTIME_DEPENDENCY_SET awm_runtime RUNTIME DESTINATION bin)
